function buildUserMessage(text) {
  const lang  = document.getElementById('output-lang').value;
  const style = document.getElementById('style-preset').value;
  const rtl   = document.getElementById('rtl-toggle').checked;
  const extra = document.getElementById('extra-instructions').value.trim();

  const styleLabels = {
    infographic: 'Business Infographic',
    report:      'Professional Report',
    dashboard:   'Data Dashboard',
    presentation:'Presentation Slides',
    magazine:    'Magazine Article'
  };

  let msg = `Convert the following document into an interactive HTML ${styleLabels[style]}.\n`;
  if (lang !== 'auto') msg += `Output language: ${lang === 'he' ? 'Hebrew (RTL)' : 'English (LTR)'}.\n`;
  if (rtl) msg += `Use RTL layout (dir="rtl").\n`;
  if (extra) msg += `Additional instructions: ${extra}\n`;
  msg += `\n--- DOCUMENT START ---\n${text}\n--- DOCUMENT END ---`;
  return msg;
}

async function callAPI(systemPrompt, userMessage, onChunk) {
  const url      = document.getElementById('api-url').value.trim();
  const apiKey   = document.getElementById('api-key').value.trim();
  const model    = document.getElementById('model-name').value.trim();
  const temp     = parseFloat(document.getElementById('temperature').value) || 0.7;
  const maxTok   = parseInt(document.getElementById('max-tokens').value) || 16000;
  const stream   = document.getElementById('streaming-toggle').checked;
  const provider = document.getElementById('provider-select').value;

  if (!url || !apiKey) throw new Error('הגדר Endpoint URL ו-API Key בהגדרות');

  const isAnthropic = provider === 'anthropic' || url.includes('anthropic.com');

  let body, headers;
  if (isAnthropic) {
    headers = {
      'Content-Type': 'application/json',
      'x-api-key': apiKey,
      'anthropic-version': '2023-06-01'
    };
    body = JSON.stringify({
      model, max_tokens: maxTok, temperature: temp,
      system: systemPrompt,
      messages: [{ role: 'user', content: userMessage }],
      stream
    });
  } else {
    headers = {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${apiKey}`
    };
    body = JSON.stringify({
      model, max_tokens: maxTok, temperature: temp,
      stream,
      messages: [
        { role: 'system', content: systemPrompt },
        { role: 'user', content: userMessage }
      ]
    });
  }

  const res = await fetch(url, { method: 'POST', headers, body });
  if (!res.ok) {
    const errText = await res.text();
    let msg = `שגיאת API ${res.status}`;
    try { const j = JSON.parse(errText); msg = j.error?.message || j.message || msg; } catch(e) {}
    throw new Error(msg);
  }

  let tokensIn = 0, tokensOut = 0;

  if (!stream) {
    const json = await res.json();
    let content = '';
    if (isAnthropic) {
      content = json.content?.[0]?.text || '';
      tokensIn  = json.usage?.input_tokens  || 0;
      tokensOut = json.usage?.output_tokens || 0;
    } else {
      content = json.choices?.[0]?.message?.content || '';
      tokensIn  = json.usage?.prompt_tokens     || 0;
      tokensOut = json.usage?.completion_tokens || 0;
    }
    onChunk(content, true);
    return { tokensIn, tokensOut };
  }

  // Streaming
  const reader = res.body.getReader();
  const dec = new TextDecoder();
  let buf = '';

  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    buf += dec.decode(value, { stream: true });
    const lines = buf.split('\n');
    buf = lines.pop();

    for (const line of lines) {
      const trimmed = line.trim();
      if (!trimmed || trimmed === 'data: [DONE]') continue;
      const dataStr = trimmed.startsWith('data: ') ? trimmed.slice(6) : trimmed;
      try {
        const json = JSON.parse(dataStr);
        let chunk = '';
        if (isAnthropic) {
          if (json.type === 'content_block_delta') chunk = json.delta?.text || '';
          if (json.type === 'message_delta') { tokensOut = json.usage?.output_tokens || tokensOut; }
          if (json.type === 'message_start')  { tokensIn  = json.message?.usage?.input_tokens || 0; }
        } else {
          chunk = json.choices?.[0]?.delta?.content || '';
          if (json.usage) {
            tokensIn  = json.usage.prompt_tokens     || tokensIn;
            tokensOut = json.usage.completion_tokens || tokensOut;
          }
        }
        if (chunk) onChunk(chunk, false);
      } catch(e) { /* skip malformed */ }
    }
  }
  return { tokensIn, tokensOut };
}

async function startGeneration() {
  if (isGenerating) return;
  if (!extractedText) { setAlert('file-alert','
⚠️ בחר קובץ תחילה
'); return; }
  if (!hasApiConfig()) { setAlert('api-test-result','
⚠️ הגדר API Key ו-Endpoint
'); return; }

  isGenerating = true;
  generatedHTML = '';
  const startTime = Date.now();
  let firstTokenTime = null;

  const btn = document.getElementById('generate-btn');
  btn.disabled = true;
  document.getElementById('gen-btn-icon').textContent = '⏳';
  document.getElementById('gen-btn-text').textContent = 'מייצר...';
  document.getElementById('progress-wrap').style.display = 'block';
  document.getElementById('empty-state').style.display = 'none';
  document.getElementById('output-iframe').style.display = 'none';
  document.getElementById('source-view').style.display = 'none';
  document.getElementById('copy-btn').disabled = true;
  document.getElementById('download-btn').disabled = true;
  switchTab('preview');
  setProgress(5, 'שולח בקשה ל-API...');
  setStatus('running', 'יוצר HTML אינטראקטיבי...');

  const sysPrompt  = document.getElementById('system-prompt').value;
  const userMsg    = buildUserMessage(extractedText);

  let htmlBuffer = '';
  let chunkCount = 0;

  try {
    const result = await callAPI(sysPrompt, userMsg, (chunk, isFinal) => {
      if (!firstTokenTime && chunk) {
        firstTokenTime = Date.now() - startTime;
        setProgress(15, 'מקבל תגובה...');
      }
      htmlBuffer += chunk;
      chunkCount++;

      // Update progress estimate
      const progress = Math.min(90, 15 + (htmlBuffer.length / 15000) * 75);
      setProgress(progress, `מקבל HTML... (${htmlBuffer.length.toLocaleString()} תווים)`);

      // Live preview every ~50 chunks
      if (chunkCount % 50 === 0 || isFinal) {
        previewPartial(htmlBuffer);
      }
    });

    generatedHTML = cleanHTML(htmlBuffer);
    const elapsed = Date.now() - startTime;

    setProgress(100, 'הושלם!');
    showResult(generatedHTML);
    setStatus('done', `✅ הושלם תוך ${(elapsed/1000).toFixed(1)}ש׳ | ${result.tokensIn.toLocaleString()} tokens in, ${result.tokensOut.toLocaleString()} tokens out`);
    updateMetrics(result.tokensIn, result.tokensOut, elapsed, generatedHTML.length, firstTokenTime, startTime);

  } catch (err) {
    setStatus('error', `❌ שגיאה: ${err.message}`);
    setProgress(0, '');
    document.getElementById('progress-wrap').style.display = 'none';
    document.getElementById('empty-state').style.display = 'flex';
    document.getElementById('empty-state').innerHTML = `
      
❌

      
${err.message}

      
בדוק את הגדרות ה-API ונסה שוב
`;
  } finally {
    isGenerating = false;
    btn.disabled = false;
    document.getElementById('gen-btn-icon').textContent = '✨';
    document.getElementById('gen-btn-text').textContent = 'צור מסמך HTML אינטראקטיבי';
    setTimeout(() => { document.getElementById('progress-wrap').style.display = 'none'; }, 2000);
  }
}

function cleanHTML(raw) {
  let html = raw.trim();
  if (html.startsWith('```html')) html = html.slice(7);
  else if (html.startsWith('```'))  html = html.slice(3);
  if (html.endsWith('```')) html = html.slice(0, -3);
  html = html.trim();
  return injectOfflineAssets(html);
}
